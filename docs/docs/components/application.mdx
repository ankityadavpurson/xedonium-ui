Building blocks for full app screens.

## Navbar

Plain horizontal nav (no sticky positioning or theme toggle; use `AppBar` for that). Supports `linkComponent` /
`linkProp` for routers.

```jsx live
<Navbar
	brand="Acme"
	links={[
		{ href: '#', label: 'Home', active: true },
		{ href: '#docs', label: 'Docs' },
	]}
	actions={<Button variant="secondary">Sign in</Button>}
/>
```

## Sidebar

`items: [{ key, label, icon?, href?, onClick?, badge?, children? }]`. `collapsed` shows icons only.

```jsx live
function Demo() {
	const [active, setActive] = useState('home')
	const [collapsed, setCollapsed] = useState(false)
	return (
		<Stack gap={3}>
			<Switch label="Collapsed" checked={collapsed} onChange={setCollapsed} />
			<div className="h-80">
				<Sidebar
					collapsed={collapsed}
					activeKey={active}
					onSelect={setActive}
					header={<strong className="text-sm">Acme</strong>}
					items={[
						{ key: 'home', label: 'Home', icon: <HomeIcon /> },
						{ key: 'grid', label: 'Projects', icon: <GridIcon />, badge: 3 },
						{
							key: 'settings',
							label: 'Settings',
							icon: <SettingsIcon />,
							children: [
								{ key: 'profile', label: 'Profile' },
								{ key: 'security', label: 'Security' },
							],
						},
					]}
				/>
			</div>
		</Stack>
	)
}
```

## AppShell

Header on top, `sidebar` on the left, scrolling main area. Below the `md` breakpoint the sidebar moves into a drawer
behind a menu button. `sidebar` can be a function `(close) => node` to close the drawer on selection. The shell fills
the viewport height; size it with `className` when embedding.

```jsx live
function Demo() {
	const [active, setActive] = useState('home')
	return (
		<AppShell
			className="!h-96 border border-app-border"
			header={<strong>Acme</strong>}
			sidebar={close => (
				<Sidebar
					activeKey={active}
					onSelect={key => {
						setActive(key)
						close()
					}}
					items={[
						{ key: 'home', label: 'Home', icon: <HomeIcon /> },
						{ key: 'settings', label: 'Settings', icon: <SettingsIcon /> },
					]}
				/>
			)}
		>
			<p>Showing: {active}</p>
		</AppShell>
	)
}
```

## CommandPalette

Open it from a shortcut with [`useKeyboardShortcuts`](../hooks). Arrow keys move, Enter runs, Escape closes.

```jsx live
function Demo() {
	const [open, setOpen] = useState(false)
	const [last, setLast] = useState('none')
	useKeyboardShortcuts({ 'mod+k': () => setOpen(true) })
	return (
		<>
			<Button onClick={() => setOpen(true)}>Open palette (Ctrl/Cmd+K)</Button>
			<p>Last command: {last}</p>
			<CommandPalette
				open={open}
				onClose={() => setOpen(false)}
				commands={[
					{ key: 'new', label: 'New project', group: 'Create', shortcut: 'N', onSelect: () => setLast('new') },
					{ key: 'invite', label: 'Invite teammate', group: 'Create', onSelect: () => setLast('invite') },
					{ key: 'theme', label: 'Toggle theme', description: 'Switch light / dark', group: 'View', onSelect: () => setLast('theme') },
				]}
			/>
		</>
	)
}
```

## NotificationCenter

```jsx live
function Demo() {
	const [items, setItems] = useState([
		{ id: 1, title: 'Build finished', body: 'main passed all checks', time: '2 min ago' },
		{ id: 2, title: 'New comment', body: 'Ada replied to your review', time: '1 h ago' },
		{ id: 3, title: 'Welcome', time: 'Yesterday', read: true },
	])
	return (
		<NotificationCenter
			notifications={items}
			onMarkRead={id => setItems(list => list.map(n => (n.id === id ? { ...n, read: true } : n)))}
			onMarkAllRead={() => setItems(list => list.map(n => ({ ...n, read: true })))}
			onClear={() => setItems([])}
		/>
	)
}
```
