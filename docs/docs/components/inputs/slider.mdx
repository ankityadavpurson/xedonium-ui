`min`, `max`, `step` and `showValue`; `onChange` receives a number.

```jsx live
function Demo() {
	const [volume, setVolume] = useState(40)
	return (
		<div className="max-w-sm">
			<Slider label="Volume" value={volume} onChange={setVolume} />
		</div>
	)
}
```
