title: "ActionMenu"
sidebar_position: 2
---

`items: [{ key, label, description?, badge?, tone?, disabled?, hasDialog?, onClick }]`. Closes on outside click and
Escape.

```jsx live
function Demo() {
	const { toast, showToast } = useTimedToast()
	return (
		<>
			<ActionMenu
				label="Actions"
				trigger="Actions"
				items={[
					{ key: 'a', label: 'First', description: 'Does a thing', onClick: () => showToast('First') },
					{ key: 'b', label: 'Second', badge: 3, onClick: () => showToast('Second', 'error') },
				]}
			/>
			<Toast toast={toast} />
		</>
	)
}
```
